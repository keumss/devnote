이 노트에서는 인증된 작성자가 사진이 포함된 블로그 포스트를 발행하고, 자신의 포스트만 수정·삭제하는 완성된 API를 만듭니다. 파일 업로드, 포스트와 사진 관계 저장, 검색, 페이지네이션, 통합 테스트까지 하나의 실행 가능한 프로젝트로 연결합니다. 첫 토픽의 구조와 설정을 준비한 뒤, 뒤의 코드를 순서대로 추가합니다.

## 프로젝트 구조와 모델·설정 준비하기

완성된 API의 파일 구조를 먼저 정리합니다. 관심사별로 파일을 나눠 라우터, 모델, 인증, 데이터베이스를 분리합니다.

```text
blog/
├── main.py           # app, lifespan, 라우터 연결, StaticFiles
├── database.py       # engine, get_session, SessionDep
├── models.py         # Author, BlogPost, PostImage
├── schemas.py        # 입출력 모델
├── auth.py           # get_current_author, CurrentAuthor
├── routers/
│   ├── posts.py      # 포스트 발행·수정·삭제·검색
│   └── images.py     # 이미지 업로드·조회
└── uploads/          # 업로드된 파일
```

`database.py`에서 엔진과 세션 의존성을 선언합니다.

```python
# database.py
from typing import Annotated

from fastapi import Depends
from sqlmodel import Session, create_engine

engine = create_engine('sqlite:///blog.db', connect_args={'check_same_thread': False})


def get_session():
    with Session(engine) as session:
        yield session


SessionDep = Annotated[Session, Depends(get_session)]
```

`models.py`에서 세 테이블을 선언합니다. Part 2에서 만든 관계 구조를 그대로 사용합니다.

```python
# models.py
from sqlmodel import Field, Relationship, SQLModel


class Author(SQLModel, table=True):
    id: int | None = Field(default=None, primary_key=True)
    name: str = Field(index=True)
    posts: list['BlogPost'] = Relationship(back_populates='author')


class BlogPost(SQLModel, table=True):
    id: int | None = Field(default=None, primary_key=True)
    title: str = Field(index=True)
    content: str
    thumbnail_url: str | None = None
    author_id: int = Field(foreign_key='author.id', index=True)
    author: Author = Relationship(back_populates='posts')
    images: list['PostImage'] = Relationship(back_populates='post')


class PostImage(SQLModel, table=True):
    id: int | None = Field(default=None, primary_key=True)
    url: str
    alt_text: str
    position: int
    post_id: int = Field(foreign_key='blogpost.id', index=True)
    post: BlogPost = Relationship(back_populates='images')
```

`schemas.py`에서 입출력 모델을 선언합니다.

```python
# schemas.py
from sqlmodel import Field, SQLModel


class PostImageInput(SQLModel):
    url: str
    alt_text: str


class BlogPostPublish(SQLModel):
    title: str
    content: str
    thumbnail_url: str | None = None
    images: list[PostImageInput] = Field(default_factory=list)


class BlogPostPatch(SQLModel):
    title: str | None = None
    content: str | None = None
    thumbnail_url: str | None = None


class AuthorPublic(SQLModel):
    id: int
    name: str


class PostImagePublic(SQLModel):
    id: int
    url: str
    alt_text: str
    position: int


class BlogPostListItem(SQLModel):
    id: int
    title: str
    thumbnail_url: str | None
    author: AuthorPublic


class BlogPostDetail(BlogPostListItem):
    content: str
    images: list[PostImagePublic]
```

`auth.py`에서 Part 3의 현재 작성자 의존성을 가져옵니다.

```python
# auth.py
from typing import Annotated

from fastapi import Depends, HTTPException, Request
from sqlmodel import Session

from database import SessionDep
from models import Author


def get_current_author(request: Request, session: SessionDep) -> Author:
    author_id_text = request.headers.get('X-Author-Id')
    if author_id_text is None or not author_id_text.isdecimal():
        raise HTTPException(status_code=401, detail='로그인이 필요합니다.')
    author = session.get(Author, int(author_id_text))
    if author is None:
        raise HTTPException(status_code=401, detail='작성자를 찾을 수 없습니다.')
    return author


CurrentAuthor = Annotated[Author, Depends(get_current_author)]
```

`main.py`에서 앱을 구성하고 라우터를 연결합니다.

```python
# main.py
from contextlib import asynccontextmanager

from fastapi import FastAPI
from fastapi.staticfiles import StaticFiles
from sqlmodel import SQLModel

from database import engine
from routers import images, posts


@asynccontextmanager
async def lifespan(app: FastAPI):
    SQLModel.metadata.create_all(engine)
    yield


app = FastAPI(lifespan=lifespan)
app.mount('/static/uploads', StaticFiles(directory='uploads'), name='uploads')
app.include_router(posts.router)
app.include_router(images.router)
```

## 파일 업로드와 포스트 발행을 트랜잭션으로 묶기

`POST /posts`에서 포스트와 사진을 한 트랜잭션에 저장합니다. `read_detail()` 헬퍼로 발행 응답과 조회 응답을 통일합니다. 아래 코드는 `routers/posts.py`에 작성합니다.

```python
# routers/posts.py
from fastapi import APIRouter, HTTPException, Query
from sqlalchemy.orm import selectinload
from sqlmodel import Session, select

from auth import CurrentAuthor
from database import SessionDep
from models import Author, BlogPost, PostImage
from schemas import (
    AuthorPublic,
    BlogPostDetail,
    BlogPostListItem,
    BlogPostPatch,
    BlogPostPublish,
    PostImagePublic,
)

router = APIRouter(prefix='/posts', tags=['posts'])


def read_detail(post_id: int, session: Session) -> BlogPostDetail:
    post = session.get(BlogPost, post_id)
    if post is None:
        raise HTTPException(status_code=404, detail='포스트를 찾을 수 없습니다.')
    images = session.exec(
        select(PostImage).where(PostImage.post_id == post_id).order_by(PostImage.position)
    ).all()
    return BlogPostDetail(
        id=post.id,
        title=post.title,
        content=post.content,
        thumbnail_url=post.thumbnail_url,
        author=AuthorPublic.model_validate(post.author),
        images=[PostImagePublic.model_validate(img) for img in images],
    )


@router.post('/', response_model=BlogPostDetail, status_code=201)
def publish_post(
    data: BlogPostPublish,
    current_author: CurrentAuthor,
    session: SessionDep,
) -> BlogPostDetail:
    try:
        post = BlogPost(
            title=data.title,
            content=data.content,
            thumbnail_url=data.thumbnail_url,
            author_id=current_author.id,
        )
        session.add(post)
        session.flush()
        session.add_all([
            PostImage(post_id=post.id, url=img.url, alt_text=img.alt_text, position=idx)
            for idx, img in enumerate(data.images, start=1)
        ])
        session.commit()
    except Exception:
        session.rollback()
        raise
    session.refresh(post)
    return read_detail(post.id, session)


@router.get('/{post_id}', response_model=BlogPostDetail)
def get_post(post_id: int, session: SessionDep) -> BlogPostDetail:
    return read_detail(post_id, session)
```

`flush()`는 트랜잭션을 확정하지 않고 새 포스트의 ID를 데이터베이스에 반영해 사진의 `post_id`에 쓸 수 있게 합니다. `read_detail()`이 한 곳에 있으므로 응답 필드를 추가하거나 사진 정렬 규칙을 바꿀 때 발행 응답과 조회 응답이 어긋나지 않습니다.

## 소유자 확인으로 수정·삭제 API 완성하기

발행한 작성자만 자신의 포스트를 수정·삭제할 수 있도록 Part 3의 소유자 확인 패턴을 적용합니다. 같은 `routers/posts.py`에 추가합니다.

```python
def get_owned_post(post_id: int, author_id: int, session: Session) -> BlogPost:
    post = session.get(BlogPost, post_id)
    if post is None:
        raise HTTPException(status_code=404, detail='포스트를 찾을 수 없습니다.')
    if post.author_id != author_id:
        raise HTTPException(status_code=403, detail='수정 권한이 없습니다.')
    return post


@router.patch('/{post_id}', response_model=BlogPostDetail)
def patch_post(
    post_id: int,
    data: BlogPostPatch,
    current_author: CurrentAuthor,
    session: SessionDep,
) -> BlogPostDetail:
    post = get_owned_post(post_id, current_author.id, session)
    post.sqlmodel_update(data.model_dump(exclude_unset=True))
    session.add(post)
    session.commit()
    return read_detail(post_id, session)


@router.delete('/{post_id}', status_code=204)
def delete_post(
    post_id: int,
    current_author: CurrentAuthor,
    session: SessionDep,
) -> None:
    post = get_owned_post(post_id, current_author.id, session)
    session.delete(post)
    session.commit()
```

사진 교체도 같은 소유자 확인 뒤 `Session.begin()` 안에서 기존 `PostImage`를 삭제하고 새 배열을 저장하는 방식으로 구현합니다. Part 3의 `replace_images()` 패턴을 그대로 적용합니다.

## 검색 API와 페이지네이션 의존성 만들기

제목 검색과 페이지네이션을 추가합니다. 페이지네이션을 의존성으로 분리하면 여러 목록 API에서 재사용할 수 있습니다.

```python
from typing import Annotated

from fastapi import Depends


class Pagination(SQLModel):
    offset: int = 0
    limit: int = 20


def get_pagination(
    offset: int = 0,
    limit: int = Query(default=20, ge=1, le=100),
) -> Pagination:
    return Pagination(offset=offset, limit=limit)


PaginationDep = Annotated[Pagination, Depends(get_pagination)]


@router.get('/', response_model=list[BlogPostListItem])
def list_posts(session: SessionDep, page: PaginationDep) -> list[BlogPost]:
    statement = (
        select(BlogPost)
        .options(selectinload(BlogPost.author))
        .order_by(BlogPost.id.desc())
        .offset(page.offset)
        .limit(page.limit)
    )
    return session.exec(statement).all()


@router.get('/search', response_model=list[BlogPostListItem])
def search_posts(
    session: SessionDep,
    q: str = Query(min_length=1, max_length=100),
) -> list[BlogPost]:
    pattern = f'%{q.strip()}%'
    statement = (
        select(BlogPost)
        .options(selectinload(BlogPost.author))
        .where(BlogPost.title.ilike(pattern))
        .order_by(BlogPost.id.desc())
        .limit(20)
    )
    return session.exec(statement).all()
```

빈 검색어는 `Query(min_length=1)`로 요청 단계에서 거절합니다. `ilike()`의 대소문자 처리는 데이터베이스마다 다를 수 있으며, 검색 규모가 커지면 전문 검색(Full-Text Search) 기능과 인덱스를 검토합니다.

## 통합 테스트로 발행·수정·권한 흐름 검증하기

발행 API가 작성자 ID, 사진 순서, 권한 거부를 모두 지키는지 API 수준에서 테스트합니다. `get_session`을 메모리 데이터베이스로 교체합니다.

```python
from fastapi.testclient import TestClient
from sqlalchemy.pool import StaticPool
from sqlmodel import Session, SQLModel, create_engine

from database import get_session
from main import app
from models import Author

test_engine = create_engine(
    'sqlite://',
    connect_args={'check_same_thread': False},
    poolclass=StaticPool,
)
SQLModel.metadata.create_all(test_engine)


def get_test_session():
    with Session(test_engine) as session:
        yield session


app.dependency_overrides[get_session] = get_test_session
client = TestClient(app)


def test_publish_assigns_author_and_photo_order() -> None:
    with Session(test_engine) as session:
        author = Author(name='민지')
        session.add(author)
        session.commit()
        session.refresh(author)

    response = client.post(
        '/posts/',
        headers={'X-Author-Id': str(author.id)},
        json={
            'title': '봄 산책',
            'content': '꽃이 피었습니다.',
            'images': [
                {'url': '/static/uploads/spring-1.jpg', 'alt_text': '벚꽃'},
                {'url': '/static/uploads/spring-2.jpg', 'alt_text': '산책로'},
            ],
        },
    )
    assert response.status_code == 201
    body = response.json()
    assert body['author']['id'] == author.id
    assert [img['position'] for img in body['images']] == [1, 2]


def test_patch_rejects_another_authors_post() -> None:
    with Session(test_engine) as session:
        owner = Author(name='하은')
        other = Author(name='도윤')
        session.add_all([owner, other])
        session.commit()
        session.refresh(owner)
        session.refresh(other)

    publish = client.post(
        '/posts/',
        headers={'X-Author-Id': str(owner.id)},
        json={'title': '비공개 초안', 'content': '본문'},
    )
    post_id = publish.json()['id']

    response = client.patch(
        f'/posts/{post_id}',
        json={'title': '바꿀 수 없는 제목'},
        headers={'X-Author-Id': str(other.id)},
    )
    assert response.status_code == 403


def test_search_finds_matching_posts() -> None:
    response = client.get('/posts/search', params={'q': '봄'})
    assert response.status_code == 200
    titles = [post['title'] for post in response.json()]
    assert any('봄' in title for title in titles)
```

테스트마다 작성자와 데이터를 명시적으로 준비하면 이전 테스트의 포스트가 결과에 영향을 주지 않습니다. 업로드 저장소를 붙일 때도 저장소 클라이언트를 의존성으로 분리해 성공·실패를 독립적으로 시험합니다.
